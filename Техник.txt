►27.10◄


Ростовое изображение - изображение построенное из маленьких пикселей. При увеличение становится размытым или зернистым, ухудшается качество

Векторное изображение - построенное из геометрических фигур, собирается формулами. При изменение масштаба - качество не изменяется

Оцифровка -  обработка инфо (сканер, МФУ, фотокамера)
Обработка - кадр, размер, цвет, формат
Публикация - сайт, медиабаза, каталог

Разрешение - число точек на дюйм (DPI)

глубина ывета - чено-белые, типографии, RGB цвета





(телефон)



struct Contact 
{
  char name[20];
  char phone[16];
};

Contact contacts[10];
int contactCount = 0;

String readLine() 
{
  while (Serial.available() == 0) {} 
  return Serial.readStringUntil('\n');
}

void showMenu() 
{
  Serial.println("1 - dobavit, 2 - naiti, 3 - vse, 0 - vihod");
}

void addContact() 
{
  if (contactCount >= 10) 
  {
    Serial.println("Spisok polon!");
    return;
  }
  Serial.print("Imya: ");
  String name = readLine();
  if (name.length() == 0) 
  {
    Serial.println("Imya ne mozhet byt pustym.");
    return;
  }
  Serial.print("Telefon: ");
  String phone = readLine();
  if (phone.length() == 0) 
  {
    Serial.println("Telefon ne mozhet byt pustym.");
    return;
  }
  name.toCharArray(contacts[contactCount].name, 20);
  phone.toCharArray(contacts[contactCount].phone, 16);
  contactCount++;
  Serial.println("Kontakt dobavlen.");
}

void findContact() 
{
  Serial.print("Imya: ");
  String name = readLine();
  for (int i = 0; i < contactCount; i++) 
  {
    if (name.equals(contacts[i].name)) 
    {
      Serial.print(contacts[i].name);
      Serial.print(": ");
      Serial.println(contacts[i].phone);
      return;
    }
  }
  Serial.println("Ne nayden.");
}

void showAll() 
{
  if (contactCount == 0) 
  {
    Serial.println("Spisok kontaktov pust.");
    return;
  }
  for (int i = 0; i < contactCount; i++) 
  {
    Serial.print(contacts[i].name);
    Serial.print(": ");
    Serial.println(contacts[i].phone);
  }
}

void setup() 
{
  Serial.begin(9600);
}

void loop() 
{
  showMenu();
  String choice = readLine();

  if (choice == "1")      addContact();
  else if (choice == "2") findContact();
  else if (choice == "3") showAll();
  else if (choice == "0") Serial.println("Poka!");
  else                    Serial.println("Ne ponyal.");
}
