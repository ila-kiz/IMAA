#include <Wire.h>
#include <Adafruit_BMP085.h>
Adafruit_BMP085 sensor;
bool ready = false;
const int BUTTON_PIN = 8; 
const int  LED_PIN = 13;
void setup() {
    pinMode(BUTTON_PIN, INPUT_PULLUP);
    pinMode(LED_PIN, OUTPUT);
    Serial.begin(9600);
    ready = sensor.begin(9600);
    if(!ready){Serial.println("SENSOR NOT FOUND");}

}

void loop() {
  if(ready){
    int state = digitalRead(BUTTON_PIN);
    if (state == LOW){digitalWrite(LED_PIN, HIGH);} else {digitalWrite(LED_PIN, LOW);} 
    delay(1000);

    float temp = sensor.readTemperature(); 
    Serial.print("T= "); 
    Serial.print(temp, 1); 
    Serial.print("C\n");
    } 
    delay(1000);

}
